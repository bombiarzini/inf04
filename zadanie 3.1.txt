using System;

public class HelloWorld
{
    public static void Main(string[] args)
    {
        Console.WriteLine("Podaj liczbe punktow od 0-100");
        string Punkty=Console.ReadLine();
        int punkty=int.Parse(Punkty);

        if(punkty>0 && punkty<=50)
        {
        Console.WriteLine("Ocena niedostaczna");
        }
        else if(punkty>51 && punkty<=74)
        {
        Console.WriteLine("Ocena dostateczna"); 
        }
        else if(punkty>75 && punkty<=89)
        {
        Console.WriteLine("Ocena dobra");
        }
        else
        {
        Console.WriteLine("Ocena bardzo dobra");
        }
            
    }
}
